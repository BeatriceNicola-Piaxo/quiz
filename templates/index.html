from flask import Flask, render_template, request, jsonify import json import os from datetime import datetime, timezone from pathlib import Path BASE_DIR = Path(__file__).resolve().parent RESULTS_FILE = BASE_DIR / "results.json" app = Flask(__name__) @app.get("/") def index(): return render_template("index.html") @app.get("/health") def health(): return jsonify({"status": "ok"}) @app.get("/results") def get_results(): if not RESULTS_FILE.exists(): return jsonify([]) try: with RESULTS_FILE.open("r", encoding="utf-8") as file: results = json.load(file) return jsonify(results) except (json.JSONDecodeError, OSError): return jsonify([]) @app.post("/submit") def submit_score(): payload = request.get_json(silent=True) or {} name = str(payload.get("name") or "").strip() q1 = str(payload.get("q1") or "").strip() q2 = str(payload.get("q2") or "").strip() score = str(payload.get("score") or "").strip() if not name or not q1 or not q2 or not score: return jsonify({"error": "Missing required fields: name, q1, q2 and score"}), 400 results = [] if RESULTS_FILE.exists(): try: with RESULTS_FILE.open("r", encoding="utf-8") as file: existing_results = json.load(file) if isinstance(existing_results, list): results = existing_results except (json.JSONDecodeError, OSError): results = [] entry = { "name": name, "score": score, "answers": {"q1": q1, "q2": q2}, "submitted_at": datetime.now(timezone.utc).isoformat() } results.append(entry) with RESULTS_FILE.open("w", encoding="utf-8") as file: json.dump(results, file, ensure_ascii=False, indent=2) return jsonify({ "message": "Score saved successfully", "name": name, "score": score, "answers": {"q1": q1, "q2": q2} }) if __name__ == "__main__": app.run(host="0.0.0.0", port=int(os.environ.get("PORT", 5000)))
